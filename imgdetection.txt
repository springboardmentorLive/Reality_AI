import torch
import torchvision
from torchvision.transforms import functional as F
from torchvision.models.detection import FasterRCNN_ResNet50_FPN_Weights
import cv2

weights = FasterRCNN_ResNet50_FPN_Weights.DEFAULT
model = torchvision.models.detection.fasterrcnn_resnet50_fpn(weights=weights)
model.eval()

labels = weights.meta["categories"]

img_path = "toy.jpg"
image = cv2.imread(img_path)
orig = image.copy()

image_rgb = cv2.cvtColor(image, cv2.COLOR_BGR2RGB)
tensor = F.to_tensor(image_rgb)
tensor = tensor.unsqueeze(0)

with torch.no_grad():
    predictions = model(tensor)

boxes = predictions[0]["boxes"]
scores = predictions[0]["scores"]
classes = predictions[0]["labels"]

for box, score, cls in zip(boxes, scores, classes):
    if score < 0.5:  # confidence threshold
        continue

    x1, y1, x2, y2 = box.int().tolist()
    label = f"{labels[cls]}: {score:.2f}"

    cv2.rectangle(orig, (x1, y1), (x2, y2), (0, 255, 0), 2)
    cv2.putText(orig, label, (x1, y1 - 5),
                cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 255, 0), 2)

import torch

from torchvision.models.detection import FasterRCNN_ResNet50_FPN_Weights

model = torchvision.models.detection.fasterrcnn_resnet50_fpn(weights=weights)
model.eval()

labels = weights.meta["categories"]

img_path = "grp.jpeg"
image = cv2.imread(img_path)
orig = image.copy()

image_rgb = cv2.cvtColor(image, cv2.COLOR_BGR2RGB)
tensor = F.to_tensor(image_rgb).unsqueeze(0)

with torch.no_grad():
    predictions = model(tensor)

boxes = predictions[0]["boxes"]
scores = predictions[0]["scores"]
classes = predictions[0]["labels"]

person_count = 0

for box, score, cls in zip(boxes, scores, classes):

    if score < 0.5:
        continue

    if labels[cls] != "person":
        continue

    person_count += 1

    x1, y1, x2, y2 = box.int().tolist()
    label = f"Person: {score:.2f}"

    cv2.rectangle(orig, (x1, y1), (x2, y2), (0, 255, 0), 2)
    cv2.putText(orig, label, (x1, y1 - 5),
                cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 255, 0), 2)

print("Total Persons Detected:", person_count)

cv2.putText(orig, f"Persons: {person_count}", (20, 40),

import torch
import torchvision
from torchvision.transforms import functional as F
from torchvision.models.detection import FasterRCNN_ResNet50_FPN_Weights
import torchvision.ops as ops
import cv2
from google.colab.patches import cv2_imshow

weights = FasterRCNN_ResNet50_FPN_Weights.DEFAULT
model = torchvision.models.detection.fasterrcnn_resnet50_fpn(weights=weights)
model.eval()

labels = weights.meta["categories"]

img_path = "grp.jpeg"
image = cv2.imread(img_path)

if image is None:
    raise ValueError("Image not found. Check the name with !ls")

orig = image.copy()

image_rgb = cv2.cvtColor(image, cv2.COLOR_BGR2RGB)
tensor = F.to_tensor(image_rgb).unsqueeze(0)

with torch.no_grad():
    predictions = model(tensor)

boxes = predictions[0]["boxes"]
scores = predictions[0]["scores"]
classes = predictions[0]["labels"]

keep = ops.nms(boxes, scores, iou_threshold=0.4)

boxes = boxes[keep]
scores = scores[keep]
classes = classes[keep]

person_count = 0

for box, score, cls in zip(boxes, scores, classes):

    if score < 0.5:
        continue

    if labels[cls] != "person":
        continue

    person_count += 1

    x1, y1, x2, y2 = box.int().tolist()
    label = f"Person: {score:.2f}"

    cv2.rectangle(orig, (x1, y1), (x2, y2), (0, 255, 0), 2)
    cv2.putText(orig, label, (x1, y1 - 5),
                cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 255, 0), 2)

print("Total Persons Detected:", person_count)

cv2.putText(orig, f"Persons: {person_count}", (20, 40),
            cv2.FONT_HERSHEY_SIMPLEX, 1, (0, 0, 255), 2)

cv2_imshow(orig)
